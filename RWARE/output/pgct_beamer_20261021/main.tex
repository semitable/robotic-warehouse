\begin{frame}{Problem Setting}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Situation}
        \begin{itemize}
          \item Two teams share one warehouse.
          \item Each agent knows its own goal, but not who its teammates are.
          \item Sharing critics with the wrong team biases the critic toward the other team's goal.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{What can be exchanged}
        Neighbours $j \in \mathcal{N}_i(t)$ (inside communication range) exchange only network outputs on probe inputs.

        \medskip
        Not exchanged: observations, rewards, parameters, team labels.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Problem Setting}
  \begin{block}{Definition 1 (Local team relation)}
    \[
      \mathcal{T}_i(t) = \{\, j \in \mathcal{N}_i(t) : z_j = z_i \,\}
    \]
    A method recovers the local team relation at time $t$ if $\hat{\mathcal{T}}_i(t) = \mathcal{T}_i(t)$ for every agent $i$.
  \end{block}

  \begin{block}{Three questions}
    \begin{itemize}
      \item Q1: Can an agent identify its teammates from action distributions only?
      \item Q2: Does a teammate's critic actually help learning?
      \item Q3: Does a correct answer to Q1 lead to faster learning?
    \end{itemize}
  \end{block}
\end{frame}

% ------------------------------------------------------------
\begin{frame}{PGCT: Main Idea}
  \begin{block}{Premise}
    Agents with the same goal act similarly in the same situation.
  \end{block}

  \begin{enumerate}
    \item \textbf{Who is nearby?} Compare only agents inside communication range.
    \item \textbf{Ask the same questions.} Both actors answer the same probe observations $q_1,\dots,q_M$.
    \item \textbf{Gate on the divergence.} Small smoothed JS divergence $\Rightarrow$ treat $j$ as a teammate.
    \item \textbf{Distill the critic.} Pull $V_i$ toward $V_j$ on probe inputs. The actor learns only from its own experience.
  \end{enumerate}
\end{frame}

\begin{frame}{Probe Sets}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{What a probe is}
        \begin{itemize}
          \item One fixed observation of a small, controlled situation.
          \item Built before training from the map by a fixed rule, so every agent builds the same set.
          \item Evaluated with zero recurrent state: one forward pass, no effect on training.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Assumption 1 (Probe comparability)}
        Every probe is a valid observation for every agent and describes the situation relative to the observer's own goal in the same way.
      \end{block}
      Team-relative channels (``my requested shelf'', ``my goal'') make this hold.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Two Roles, Two Sets}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Gate set $\mathcal{Q}_{\mathrm{gate}}$}
        Needs states where the best action depends on the team.

        \medskip
        Example: carrying a requested shelf, no goal in view. The best first move depends on where my goal is.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Distillation set $\mathcal{Q}_{\mathrm{dist}}$}
        Needs wide coverage: the critic should match the teammate's critic in every task phase.

        \medskip
        Search, pickup, carry, return, choice between shelves.
      \end{block}
    \end{column}
  \end{columns}

  \medskip
  \begin{itemize}
    \item Shared setting: $\mathcal{Q}_{\mathrm{gate}} = \mathcal{Q}_{\mathrm{dist}}$ (48 probes).
    \item Split setting: $\mathcal{Q}_{\mathrm{gate}} \neq \mathcal{Q}_{\mathrm{dist}}$ (48 gate, 64 distillation).
  \end{itemize}
\end{frame}

\begin{frame}{Response Comparison}
  \small
  Let $p_i^m = \pi_i(\cdot \mid q_m, h=0)$. Every $P$ updates, neighbours exchange $\{p_i^m\}$ and compute
  \[
    D_{ij} = \frac{1}{M} \sum_{m=1}^{M} \mathrm{JS}(p_i^m, p_j^m),
  \]
  \[
    \mathrm{JS}(p,q) = \frac{1}{2}\mathrm{KL}(p \,\|\, \bar p) + \frac{1}{2}\mathrm{KL}(q \,\|\, \bar p),
    \qquad \bar p = \frac{p+q}{2}.
  \]

  \begin{block}{Why JS}
    \begin{itemize}
      \item Symmetric: $D_{ij} = D_{ji}$.
      \item Finite even when one policy gives an action probability 0.
      \item Bounded in $[0, \ln 2]$, so one threshold works for the whole run.
    \end{itemize}
  \end{block}
  Nearly deterministic policies: $D_{ij} \approx \ln 2 \times$ (fraction of probes with different actions).
\end{frame}

\begin{frame}{Gate}
  \begin{block}{Smoothing}
    The first measurement sets $\bar D_{ij} \leftarrow D_{ij}$. After that,
    \[
      \bar D_{ij} \leftarrow (1-\beta)\, \bar D_{ij} + \beta\, D_{ij}.
    \]
    Measured only while $i$ and $j$ are neighbours. $n_{ij}$ counts the measurements.
  \end{block}

  \begin{block}{Inferred teammates (after $U_0$ warm-up updates)}
    \[
      \hat{\mathcal{T}}_i(t) = \{\, j \in \mathcal{N}_i(t) : n_{ij} \ge n_{\min},\ \bar D_{ij} < \tau \,\}
    \]
  \end{block}
  \begin{itemize}
    \item Warm-up: early policies are all close to uniform and look alike.
    \item $D_{ij} = D_{ji}$ and both agents update together, so the relation is symmetric.
  \end{itemize}
\end{frame}

\begin{frame}{Critic Distillation}
  \small
  \[
    \mathcal{L}_i = \mathcal{L}^{\mathrm{PPO}}_i + \lambda \sum_{j \in \hat{\mathcal{T}}_i(t)} \alpha_{ij}\, \mathcal{L}^{D}_{i \leftarrow j},
    \qquad \alpha_{ij} = \frac{1}{|\hat{\mathcal{T}}_i(t)|}
  \]
  \[
    \mathcal{L}^{D}_{i \leftarrow j} = \frac{1}{M_d} \sum_{m=1}^{M_d} \Big( V_i(q^d_m) - \mathrm{sg}\big[V_j(q^d_m)\big] \Big)^2
  \]

  \begin{itemize}
    \item $\mathrm{sg}$: stop-gradient. $V_j$ is computed before the update and used as a fixed target.
    \item If $\hat{\mathcal{T}}_i(t) = \emptyset$, the peer term vanishes and agent $i$ is plain independent PPO.
  \end{itemize}

  \begin{block}{Why the critic, not the actor}
    \begin{itemize}
      \item The value of a state belongs to the team goal; the choice of action also depends on the agent's role.
      \item The actor is affected only through the advantage $\hat A = \hat R - V_i$.
    \end{itemize}
  \end{block}
\end{frame}

% ------------------------------------------------------------
\begin{frame}{Algorithm: PGCT for Agent $i$}
  \small
  \begin{block}{}
    \begin{enumerate}
      \item Initialize $\pi_i, V_i$. Set $\bar D_{ij}$ undefined, $n_{ij} \leftarrow 0$ for all $j$.
      \item For each update $u = 1, 2, \dots$
      \begin{enumerate}
        \item Collect a rollout with $\pi_i$; compute PPO targets from own rewards.
        \item If $u \bmod P = 0$: for each $j \in \mathcal{N}_i(t)$
        \begin{itemize}
          \item exchange $\{p^m\}$ on $\mathcal{Q}_{\mathrm{gate}}$, compute $D_{ij}$;
          \item update $\bar D_{ij}$ (first value or EMA), $n_{ij} \leftarrow n_{ij} + 1$.
        \end{itemize}
        \item If $u > U_0$: compute $\hat{\mathcal{T}}_i$; otherwise $\hat{\mathcal{T}}_i = \emptyset$.
        \item Receive pre-update $V_j(q^d_m)$ from $j \in \hat{\mathcal{T}}_i$.
        \item Update $\pi_i, V_i$ with $\mathcal{L}_i$.
      \end{enumerate}
    \end{enumerate}
  \end{block}
  Settings: $P=5$, $\beta=0.35$, $\tau=0.12$, $U_0=100$, $n_{\min}=8$, $\lambda=0.01$.
\end{frame}

\begin{frame}{When the Gate Finds Teammates}
  \small
  \begin{block}{Separation margin}
    \[
      \delta_{\mathrm{in}}(n) = \max_{z_i = z_j} D_{ij}(n), \qquad
      \delta_{\mathrm{out}}(n) = \min_{z_i \neq z_j} D_{ij}(n)
    \]
    \[
      \Delta(n) = \delta_{\mathrm{out}}(n) - \delta_{\mathrm{in}}(n)
    \]
    $\Delta > 0$: even the least similar teammate pair is closer than the most similar cross-team pair.
  \end{block}

  \begin{block}{Proposition 1 (Exact local recovery)}
    If $\Delta(n) > 0$, $|\bar D_{ij}(n) - D_{ij}(n)| \le \epsilon$ for all neighbour pairs, $2\epsilon < \Delta(n)$,
    $\delta_{\mathrm{in}} + \epsilon < \tau < \delta_{\mathrm{out}} - \epsilon$, and warm-up and $n_{\min}$ are met, then
    \[
      \hat{\mathcal{T}}_i(n) = \mathcal{T}_i(n) \quad \text{for every } i.
    \]
    No agent uses a donor from the other team.
  \end{block}
\end{frame}

\begin{frame}{Smoothing Delay}
  \begin{block}{Proposition 2 (Tracking error)}
    If consecutive measurements of a pair differ by at most $\rho$,
    \[
      |D_{ij}(n) - D_{ij}(n-1)| \le \rho,
    \]
    then the EMA started from the first measurement satisfies
    \[
      |\bar D_{ij}(n) - D_{ij}(n)| \le \frac{1-\beta}{\beta}\, \rho .
    \]
  \end{block}

  \begin{itemize}
    \item So Proposition 1 holds when $2\,\frac{1-\beta}{\beta}\,\rho < \Delta(n)$ and $\tau$ lies in the gap.
    \item $\beta = 0.35$ gives $(1-\beta)/\beta \approx 1.86$.
    \item Larger $\beta$ tracks faster but averages less.
    \item Two quantities to measure in practice: margin $\Delta$ and drift $\rho$.
  \end{itemize}
\end{frame}

\begin{frame}{When Teams Cannot Be Identified}
  \begin{block}{Proposition 3 (Teams with the same decision problem)}
    Fix the other agents' policies. If agents of different teams face the same decision problem, then a memoryless map optimal for one team is optimal for the other.

    \medskip
    If the optimal map is unique, optimal agents answer every probe the same way:
    \[
      D_{ij} = 0 \ \text{for all pairs}, \qquad \Delta = 0,
    \]
    and no threshold separates the teams.
  \end{block}

  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{E1 (goals side by side)}

      Team-relative view + symmetric map $\Rightarrow$ almost the same problem. Not expected to separate.
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{E1-F (goals in opposite corners)}

      Absolute position in the observation + distant goals $\Rightarrow$ best action differs by team.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Summary}
  \begin{itemize}
    \item PGCT lets each agent choose whose critic to learn from, using only outputs on shared probes.
    \item The gate recovers teams when the margin $\Delta$ is larger than twice the tracking error.
    \item When both teams solve the same problem, no policy-based gate can separate them.
    \item An empty teammate set falls back to independent PPO.
  \end{itemize}
\end{frame}
